# Day 100: Create and Configure Alarm Using CloudWatch Using Terraform

# 1. First check the file 'provider.tf', if everything is fine

# 2. Update the 'main.tf' file

#File Content:

resource "aws_sns_topic" "sns_topic" {
  name = "nautilus-sns-topic"
}

resource "aws_instance" "nautilus_ec2" {
  ami           = "ami-0c02fb55956c7d316"
  instance_type = "t2.micro"

  tags = {
    Name = "nautilus-ec2"
  }
}

resource "aws_cloudwatch_metric_alarm" "nautilus_alarm" {
  alarm_name          = "nautilus-alarm"
  comparison_operator = "GreaterThanOrEqualToThreshold"
  evaluation_periods  = 1
  metric_name         = "CPUUtilization"
  namespace           = "AWS/EC2"
  period              = 300
  statistic           = "Average"
  threshold           = 90
  alarm_description   = "Alarm when CPU utilization exceeds 90%"
  alarm_actions       = [aws_sns_topic.sns_topic.arn]

  dimensions = {
    InstanceId = aws_instance.nautilus_ec2.id
  }
}

# 3. Create 'outputs.tf' file

#File Content:

output "KKE_instance_name" {
  value = aws_instance.nautilus_ec2.tags["Name"]
}

output "KKE_alarm_name" {
  value = aws_cloudwatch_metric_alarm.nautilus_alarm.alarm_name
}

# 4. Open bash terminal and run below commands to create infra

$ terraform init
$ terraform plan
$ terraform validate
$ terraform apply
